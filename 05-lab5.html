p <- ggplot(rndl, aes(x = dpw, y = Value, color = Estimator, fill = Estimator)) +
  geom_bar(stat = "identity", position = position_dodge()) +
  facet_grid(rd ~ category, scales = "free") +
  theme_bw() +                                    # Sets the black and white theme
  scale_color_brewer(palette = "Set2") +          # Colorblind-friendly color palette
  scale_fill_brewer(palette = "Set2") +           # Colorblind-friendly fill palette
  labs(title = "Species Richness and Diversity Estimates",
       x = "Days Past Weaning (dpw)",
       y = "Estimate Value")
